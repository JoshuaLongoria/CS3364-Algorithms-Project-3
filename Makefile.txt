CC ?= cc
CFLAGS ?= -std=c11 -Wall -Wextra -Wpedantic -O2
LDFLAGS ?=
LDLIBS ?=

COMMON = graph.c shortest_paths.c
HEADERS = graph.h shortest_paths.h

.PHONY: all test sanitize clean
all: routes

routes: main.c $(COMMON) $(HEADERS)
	$(CC) $(CFLAGS) main.c $(COMMON) $(LDFLAGS) $(LDLIBS) -o $@

test_routes: test_routes.c $(COMMON) $(HEADERS)
	$(CC) $(CFLAGS) -UNDEBUG test_routes.c $(COMMON) $(LDFLAGS) $(LDLIBS) -o $@

test: routes test_routes
	./test_routes

sanitize:
	$(CC) -std=c11 -Wall -Wextra -Wpedantic -g -O1 -UNDEBUG -fsanitize=address,undefined -fno-omit-frame-pointer test_routes.c $(COMMON) -o test_routes_sanitized
	./test_routes_sanitized

clean:
	rm -f routes test_routes test_routes_sanitized
